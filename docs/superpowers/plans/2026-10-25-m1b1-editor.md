```rust
//! 会话级 SQL：执行计划、读取和切换当前 schema。

use crate::{Family, ident};

/// 生成一条语句的执行计划要依次执行的语句。最后一条的结果集就是计划：
/// - MySQL 系、PG 系：`EXPLAIN <语句>`；
/// - 达梦：`EXPLAIN FOR <语句>`，返回计划表的各行（`EXPLAIN <语句>` 经 DPI 执行不返回结果集）；
/// - Oracle：`EXPLAIN PLAN FOR <语句>` 写入计划表，再用 `DBMS_XPLAN.DISPLAY` 读出文本。
///
/// 都不会真正执行语句（不用 `EXPLAIN ANALYZE`）。
pub fn explain(family: Family, sql: &str) -> Vec<String> {
    match family {
        Family::Oracle => vec![
            format!("EXPLAIN PLAN FOR {sql}"),
            "SELECT PLAN_TABLE_OUTPUT FROM TABLE(DBMS_XPLAN.DISPLAY())".to_string(),
        ],
        Family::Dameng => vec![format!("EXPLAIN FOR {sql}")],
        Family::MySql | Family::Postgres => vec![format!("EXPLAIN {sql}")],
    }
}

/// 查询当前 schema 的语句，结果是一行一列；MySQL 没有选库时为 NULL。
pub fn current_schema(family: Family) -> &'static str {
    match family {
        Family::MySql => "SELECT DATABASE()",
        Family::Postgres => "SELECT current_schema()",
        Family::Oracle | Family::Dameng => "SELECT SYS_CONTEXT('USERENV', 'CURRENT_SCHEMA') FROM DUAL",
    }
}

/// 切换当前 schema 的语句。名字总是加引号，按原样使用（不转大小写）。
pub fn set_schema(family: Family, schema: &str) -> String {
    let name = ident::quote_ident(family, schema);
    match family {
        Family::MySql => format!("USE {name}"),
        Family::Postgres => format!("SET search_path TO {name}"),
        Family::Oracle | Family::Dameng => format!("ALTER SESSION SET CURRENT_SCHEMA = {name}"),
    }
}

/// 执行后可能改变当前 schema 的语句（按首个关键字判断），执行成功后要重新读取。
pub fn may_change_schema(keyword: &str) -> bool {
    matches!(keyword, "USE" | "SET" | "ALTER")
}
```

```rust
/// `sql` 中的字节偏移（从 0 开始）换算成 `Error::Sql::position` 用的字符偏移（从 1 开始）。
/// 偏移落在字符中间时按该字符算，超出末尾时取末尾之后的位置。
pub fn char_position(sql: &str, byte: usize) -> u32 {
    let mut end = byte.min(sql.len());
    while !sql.is_char_boundary(end) {
        end -= 1;
    }
    (sql[..end].chars().count() + 1) as u32
}

impl Error {
    /// 按字节报告出错位置的数据库（如 Oracle）：把 `Sql` 错误的位置（从 1 开始的字节偏移）换成字符偏移。
    pub fn position_bytes_to_chars(self, sql: &str) -> Self {
        match self {
            Error::Sql { code, message, position: Some(p) } => Error::Sql {
                code,
                message,
                position: Some(char_position(sql, (p as usize).saturating_sub(1))),
            },
            other => other,
        }
    }
```

```rust
/// MySQL 错误码：查询被 KILL QUERY 中断。
const ER_QUERY_INTERRUPTED: u16 = 1317;

pub(crate) fn map_err(e: mysql_async::Error) -> Error {
    match e {
        mysql_async::Error::Server(ref se) => match se.code {
            ER_QUERY_INTERRUPTED => Error::Cancelled,
            // 1044 无库权限，1045 密码错误，1698 用户被拒绝
            1044 | 1045 | 1698 => Error::Auth { message: se.message.clone() },
            code => Error::Sql { code: code.to_string(), message: se.message.clone(), position: None },
        },
        mysql_async::Error::Io(_) | mysql_async::Error::Driver(_) | mysql_async::Error::Url(_) => {
            Error::connect(e.to_string())
        }
        mysql_async::Error::Other(_) => Error::internal(e.to_string()),
    }
}

/// 从语法错误信息 `… near 'FROM t' at line 2` 找出出错位置：`near` 后面是从出错处开始的原文（可能被截断），
/// 在第 2 行起找它的第一行；为空表示语句意外结束，位置在末尾。返回字节偏移。
fn locate(sql: &str, message: &str) -> Option<usize> {
    let (head, line) = message.rsplit_once("' at line ")?;
    let line: usize = line.trim().parse().ok()?;
    let near = &head[head.find(" near '")? + 7..];
    let line_start = sql.split_inclusive('\n').take(line.saturating_sub(1)).map(str::len).sum::<usize>();
    let snippet = near.split('\n').next().unwrap_or_default();
    if snippet.is_empty() {
        return Some(sql.trim_end().len());
    }
    sql.get(line_start..)?.find(snippet).map(|at| line_start + at)
}

/// 给语法错误补上出错位置（字符偏移，从 1 开始）。
pub(crate) fn with_position(e: Error, sql: &str) -> Error {
    match e {
        Error::Sql { code, message, position: None } => {
            let position = locate(sql, &message).map(|b| char_position(sql, b));
            Error::Sql { code, message, position }
        }
        other => other,
    }
}
```

```rust
//! 从达梦的错误信息中找出出错位置。达梦不单独返回位置，只在信息里写
//! `第 2 行, 第 13 列[FROM]附近出现错误`。实测列号在第一行和其他行之间差 1，不可靠，
//! 所以在该行里找方括号中的词，取离列号最近的一处；方括号为空（语句意外结束）时按列号取，超出行尾取行尾。

use dblens_core::{Error, char_position};

/// 解析 `第 L 行, 第 C 列[T]`，返回行号、列号和词。没有列号时返回 None。
fn parse(message: &str) -> Option<(usize, usize, &str)> {
    let rest = message.strip_prefix('第')?.trim_start();
    let (line, rest) = rest.split_once('行')?;
    let rest = rest.trim_start().strip_prefix(',')?.trim_start().strip_prefix('第')?.trim_start();
    let (col, rest) = rest.split_once('列')?;
    let token = rest.strip_prefix('[').and_then(|r| r.split_once(']')).map_or("", |(t, _)| t);
    Some((line.trim().parse().ok()?, col.trim().parse().ok()?, token))
}

/// 出错位置在 `sql` 中的字节偏移。
fn locate(sql: &str, message: &str) -> Option<usize> {
    let (line, col, token) = parse(message)?;
    let mut start = 0;
    for (i, text) in sql.split('\n').enumerate() {
        if i + 1 == line {
            let text = text.trim_end_matches('\r');
            let col = col.saturating_sub(1).min(text.len());
            if token.is_empty() {
                return Some(start + col);
            }
            let lower = text.to_ascii_lowercase();
            let token = token.to_ascii_lowercase();
            return lower
                .match_indices(&token)
                .map(|(at, _)| at)
                .min_by_key(|at| at.abs_diff(col))
                .map(|at| start + at);
        }
        start += text.len() + 1;
    }
    None
}

            let position = locate(sql, &message).map(|b| char_position(sql, b));
            Error::Sql { code, message, position }
        }
        other => other,
    }
}
```

**Interfaces:**

- Produces（Task 3 使用）：
  - `pub struct ExecRequest { sql: String, confirmed: bool, continue_on_error: bool, cursor: Option<usize>, explain: bool }`（`cursor` 是字节偏移）、`ExecRequest::new(sql)`；
  - `SessionManager::submit(&self, id, req: ExecRequest) -> Result<()>`：请求不合法（执行计划给了多条语句）时直接返回 `Invalid`，不产生事件。`execute(id, sql, confirmed)` 保留，内部调用 `submit`；
  - `SessionManager::set_schema(&self, id, schema: &str)`、`refresh_schema(&self, id)`；
  - `SessionInfo::schema: Option<String>`；
  - `pub fn statement_at(statements: Vec<SqlStatement>, cursor: usize) -> Option<SqlStatement>`；
  - 事件：
    - `StatementStart { index, total, start, end, info }`；
    - `ConfirmRequired { index, start, end, reason }`；
    - `Failed { index, error, offset: Option<usize> }`，`offset` 为 None 时不序列化。

行为要点：
- **光标**：落在两条语句之间时取前一条，在第一条之前时取第一条。只执行选出的那条，`index` 为 0，`start`/`end` 仍是在提交的 SQL 中的偏移。
- **执行计划**：一条语句对应方言生成的几条语句，都归在同一个 `StatementStart` 下面；按生成的语句（`EXPLAIN …`）分类，所以不会触发生产确认。
- **出错继续**：失败或被策略拒绝后继续。需要确认、被取消、连接断开时仍然停下。被拒绝的语句先发 `StatementStart` 再发 `Failed`，调用方才知道是哪一段。
- **逐条确认**：`confirmed` 只对第一条语句有效。准备阶段实测发现，原来的实现在确认 `DROP` 后，同一脚本里后面不带 WHERE 的 `DELETE` 不再询问。
- **出错位置**：`Error::Sql::position`（语句内的字符偏移）换算成提交的 SQL 中的字节偏移，放在 `Failed::offset`。执行计划的语句是生成的，位置对不上，不换算。
- **语句总数**：`total` 让前端区分最后一条。前面语句的结果没取完，后面的语句就执行不了。
- **当前 schema**：
  - 打开会话后读取一次（`Service::open_session` 调 `refresh_schema`，失败不影响打开）；
  - 执行 `USE` / `SET` / `ALTER` 开头的语句成功后重新读取；
  - 重新连接后重新读取。

```rust
    async fn next(&mut self, max_rows: usize) -> Result<Option<ExecEvent>> {
        if let Some((flag, wake)) = &self.sleep {
            while !flag.swap(false, Ordering::SeqCst) {
                wake.notified().await;
            }
            self.sleep = None;
            return Err(Error::Cancelled);
        }
        if let Some(ev) = self.events.pop_front() {
            return Ok(Some(ev));
        }
        if self.rows_left > 0 {
            let n = (max_rows as i64).min(self.rows_left);
            let rows = (0..n).map(|k| vec![Value::Int(self.next_value + k)]).collect();
            self.next_value += n;
            self.rows_left -= n;
            if self.rows_left == 0 {
                self.events.push_back(ExecEvent::ResultEnd);
            }
            return Ok(Some(ExecEvent::Rows(rows)));
        }
        Ok(None)
    }
```

```rust
//! 执行是拉取式的：调用方每调用一次 `next` 取一个事件，结果集按批读取，读多少取多少。
//! 每条语句执行前都经过 `Policy`；需要确认时脚本停下，调用方确认后用 `confirmed = true` 重新提交。

pub type SessionId = u64;

/// 执行过程中依次产生的事件，由 `SessionManager::next` 逐个取出。
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
#[serde(tag = "type", rename_all = "snake_case")]
pub enum SessionEvent {
    /// 开始执行第 `index` 条语句（从 0 开始），这次提交共 `total` 条；`start`/`end` 是它在提交的 SQL 中的字节偏移。
    /// 调用方据此区分最后一条：前面语句的结果要取完，后面的语句才能执行。
    StatementStart {
        index: usize,
        total: usize,
        start: usize,
        end: usize,
        info: StatementInfo,
    },
    ResultStart {
        columns: Vec<ColumnMeta>,
    },
    Rows {
        rows: Vec<Row>,
    },
    ResultEnd,
    Affected {
        count: u64,
    },
    Message {
        text: String,
    },
    StatementEnd {
        index: usize,
        elapsed_ms: u64,
    },
    /// 这条语句需要确认；脚本停在这里，后面的语句都没有执行。`start`/`end` 是它的字节范围。
    ConfirmRequired {
        index: usize,
        start: usize,
        end: usize,
        reason: String,
    },
    /// 这条语句失败（或被拒绝、被取消）。除非请求了 `continue_on_error`，脚本停在这里。
    /// `offset` 是数据库报告的出错位置换算成的、在提交的 SQL 中的字节偏移；没有位置时省略。
    Failed {
        index: usize,
        error: Error,
        #[serde(default, skip_serializing_if = "Option::is_none")]
        offset: Option<usize>,
    },
    /// 整个脚本结束。之后 `next` 返回 None。
    Done,
}

/// 会话的当前状态。
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct SessionInfo {
    pub id: SessionId,
    pub profile_id: String,
    pub variant: Variant,
    pub autocommit: bool,
    /// 手动提交模式下，是否有尚未提交的修改。
    pub in_transaction: bool,
    /// 是否正在执行脚本。
    pub busy: bool,
    /// 连接已断开；要先 `reconnect` 才能继续执行。
    pub broken: bool,
    /// 当前 schema（MySQL 系为当前库）；读不到时为 None。
    pub schema: Option<String>,
}

/// 一次执行请求。
#[derive(Debug, Clone, Default, PartialEq, Eq)]
pub struct ExecRequest {
    pub sql: String,
    /// 已确认第一条语句：它需要确认时直接执行。后面需要确认的语句仍会停下，每条都要单独确认。
    pub confirmed: bool,
    /// 某条语句失败或被策略拒绝后，继续执行后面的语句。需要确认、被取消、连接断开时仍然停下。
    pub continue_on_error: bool,
    /// 只执行这个字节偏移所在的那条语句：落在两条语句之间时取前一条，在第一条之前时取第一条。
    pub cursor: Option<usize>,
    /// 只生成执行计划，不执行语句。要执行的语句只能有一条（可以用 `cursor` 选出）。
    pub explain: bool,
}

impl ExecRequest {
    pub fn new(sql: impl Into<String>) -> Self {
        Self { sql: sql.into(), ..Self::default() }
    }
}

/// 光标所在的语句（规则见 `ExecRequest::cursor`）。没有语句时返回 None。
pub fn statement_at(statements: Vec<SqlStatement>, cursor: usize) -> Option<SqlStatement> {
    let first = statements.first().cloned();
    statements.into_iter().rev().find(|s| s.start <= cursor).or(first)
}

/// 一条要执行的语句。生成执行计划时 `texts` 是方言生成的几条语句，事件都归在这一条下面。
struct Step {
    index: usize,
    start: usize,
    end: usize,
    texts: Vec<String>,
}

/// 按请求选出要执行的语句。
fn plan(dialect: &Dialect, req: &ExecRequest) -> Result<Vec<Step>> {
    let mut statements = dialect.split(&req.sql);
    if let Some(cursor) = req.cursor {
        statements = statement_at(statements, cursor).into_iter().collect();
    }
    if req.explain {
        let [st] = statements.as_slice() else {
            return Err(Error::invalid("执行计划只能针对一条语句"));
        };
        return Ok(vec![Step { index: 0, start: st.start, end: st.end, texts: dialect.explain(&st.text) }]);
    }
    Ok(statements
        .into_iter()
        .enumerate()
        .map(|(index, st)| Step { index, start: st.start, end: st.end, texts: vec![st.text] })
        .collect())
}

/// 数据库报告的出错位置（语句内从 1 开始的字符偏移）换算成提交的 SQL 中的字节偏移。
/// 执行计划的语句是生成的，位置对不上编辑器，不换算。
fn error_offset(error: &Error, step: &Step) -> Option<usize> {
    let Error::Sql { position: Some(p), .. } = error else { return None };
    let [text] = step.texts.as_slice() else { return None };
    let chars = (*p as usize).checked_sub(1)?;
    let byte = text.char_indices().nth(chars).map_or(text.len(), |(b, _)| b);
    Some(step.start + byte)
}

/// 读取当前 schema：取结果第一行第一列。
async fn query_schema(conn: &mut Box<dyn Connection>, dialect: &Dialect) -> Result<Option<String>> {
    let mut stream = conn.execute(dialect.current_schema_sql(), &[]).await?;
    let mut out = None;
    while let Some(ev) = stream.next(10).await? {
        if let ExecEvent::Rows(rows) = ev
            && out.is_none()
        {
            out = rows.into_iter().next().and_then(|r| r.into_iter().next()).and_then(|v| match v {
                Value::String(s) => Some(s),
                _ => None,
            });
        }
    }
    Ok(out)
}

/// 执行一条不关心结果的语句，取完全部事件。
async fn run_quiet(conn: &mut Box<dyn Connection>, sql: &str) -> Result<()> {
    let mut stream = conn.execute(sql, &[]).await?;
    while stream.next(100).await?.is_some() {}
    Ok(())
}

enum Command {
    Execute {
        req: ExecRequest,
        reply: oneshot::Sender<Result<()>>,
    },
    /// 切换当前 schema；`None` 只重新读取。
    Schema {
        set: Option<String>,
        reply: oneshot::Sender<Result<()>>,
    },
    Next {
        max_rows: usize,
        reply: oneshot::Sender<Result<Option<SessionEvent>>>,
    },
    SetAutocommit {
        on: bool,
        reply: oneshot::Sender<Result<()>>,
    },
    Commit {
        reply: oneshot::Sender<Result<()>>,
    },
    Rollback {
        reply: oneshot::Sender<Result<()>>,
    },
    Replace {
        conn: Box<dyn Connection>,
        dialect: Dialect,
        reply: oneshot::Sender<Result<()>>,
    },
    Close {
        commit: bool,
        reply: oneshot::Sender<Result<()>>,
    },
}

struct Handle {
    tx: mpsc::Sender<Command>,
    cancel: Arc<Mutex<CancelHandle>>,
    info: Arc<Mutex<SessionInfo>>,
}

/// 管理全部会话。会话之间互不影响，每个会话的命令按顺序处理。
pub struct SessionManager {
    policy: Arc<dyn Policy>,
    metadata: Arc<MetadataCache>,
    next_id: AtomicU64,
    sessions: Mutex<HashMap<SessionId, Handle>>,
}

fn gone() -> Error {
    Error::internal("会话任务已退出")
}

impl SessionManager {
    pub fn new(policy: Arc<dyn Policy>, metadata: Arc<MetadataCache>) -> Self {
        Self { policy, metadata, next_id: AtomicU64::new(1), sessions: Mutex::new(HashMap::new()) }
    }

    fn handle<T>(&self, id: SessionId, f: impl FnOnce(&Handle) -> T) -> Result<T> {
        let sessions = self.sessions.lock().map_err(|_| Error::internal("会话表锁已损坏"))?;
        sessions.get(&id).map(f).ok_or_else(|| Error::not_found(format!("会话 {id}")))
    }

    /// 用一个已建立的连接创建会话（连接由调用方通过 `DriverRegistry::connect` 建立）。
    pub fn open(
        &self,
        profile_id: &str,
        environment: crate::profile::Environment,
        conn: Box<dyn Connection>,
        dialect: Dialect,
    ) -> Result<SessionId> {
        let id = self.next_id.fetch_add(1, Ordering::Relaxed);
        let info = Arc::new(Mutex::new(SessionInfo {
            id,
            profile_id: profile_id.to_string(),
            variant: dialect.variant(),
            autocommit: true,
            in_transaction: false,
            busy: false,
            broken: false,
            schema: None,
        }));
        let cancel = Arc::new(Mutex::new(conn.cancel_handle()));
        let (tx, rx) = mpsc::channel(16);
        let actor = Actor {
            conn: Some(conn),
            ctx: ExecContext { profile_id: profile_id.to_string(), environment, dialect },
            policy: self.policy.clone(),
            metadata: self.metadata.clone(),
            info: info.clone(),
            cancel: cancel.clone(),
            rx,
        };
        tokio::spawn(actor.run());
        self.sessions
            .lock()
            .map_err(|_| Error::internal("会话表锁已损坏"))?
            .insert(id, Handle { tx, cancel, info });
        Ok(id)
    }

    async fn call<T>(
        &self,
        id: SessionId,
        make: impl FnOnce(oneshot::Sender<Result<T>>) -> Command,
    ) -> Result<T> {
        let tx = self.handle(id, |h| h.tx.clone())?;
        let (reply, rx) = oneshot::channel();
        tx.send(make(reply)).await.map_err(|_| gone())?;
        rx.await.map_err(|_| gone())?
    }

    pub fn info(&self, id: SessionId) -> Result<SessionInfo> {
        self.handle(id, |h| {
            h.info.lock().map(|i| i.clone()).map_err(|_| Error::internal("会话状态锁已损坏"))
        })?
    }

    pub fn list(&self) -> Result<Vec<SessionInfo>> {
        let sessions = self.sessions.lock().map_err(|_| Error::internal("会话表锁已损坏"))?;
        let mut out: Vec<SessionInfo> =
            sessions.values().filter_map(|h| h.info.lock().ok().map(|i| i.clone())).collect();
        out.sort_by_key(|i| i.id);
        Ok(out)
    }

    /// 提交一段 SQL（可以有多条语句）。之后用 `next` 取事件，直到 `Done`。会话正忙时报 `Busy`。
    pub async fn execute(&self, id: SessionId, sql: &str, confirmed: bool) -> Result<()> {
        self.submit(id, ExecRequest { confirmed, ..ExecRequest::new(sql) }).await
    }

    /// 提交一次执行请求。请求不合法（如执行计划给了多条语句）时直接报错，不产生事件。
    pub async fn submit(&self, id: SessionId, req: ExecRequest) -> Result<()> {
        self.call(id, |reply| Command::Execute { req, reply }).await
    }

    /// 切换当前 schema，成功后会话信息中的 `schema` 随之更新。
    pub async fn set_schema(&self, id: SessionId, schema: &str) -> Result<()> {
        let set = Some(schema.to_string());
        self.call(id, |reply| Command::Schema { set, reply }).await
    }

    /// 重新读取当前 schema。
    pub async fn refresh_schema(&self, id: SessionId) -> Result<()> {
        self.call(id, |reply| Command::Schema { set: None, reply }).await
    }

    /// 取下一个事件；`max_rows` 是一个 `Rows` 事件最多包含的行数。没有在执行的脚本时返回 None。
    pub async fn next(&self, id: SessionId, max_rows: usize) -> Result<Option<SessionEvent>> {
        self.call(id, |reply| Command::Next { max_rows: max_rows.max(1), reply }).await
    }

    /// 取消正在执行的语句。可以在会话忙碌时调用；没有在执行时什么也不做。
    pub async fn cancel(&self, id: SessionId) -> Result<()> {
        let cancel = self.handle(id, |h| h.cancel.clone())?;
        let handle = cancel.lock().map_err(|_| Error::internal("取消句柄锁已损坏"))?.clone();
        handle.cancel().await
    }

    pub async fn set_autocommit(&self, id: SessionId, on: bool) -> Result<()> {
        self.call(id, |reply| Command::SetAutocommit { on, reply }).await
    }

    pub async fn commit(&self, id: SessionId) -> Result<()> {
        self.call(id, |reply| Command::Commit { reply }).await
    }

    pub async fn rollback(&self, id: SessionId) -> Result<()> {
        self.call(id, |reply| Command::Rollback { reply }).await
    }

    /// 换上一个新建立的连接（连接断开后由用户确认重连）。原连接上未提交的修改已经丢失。
    pub async fn replace_connection(
        &self,
        id: SessionId,
        conn: Box<dyn Connection>,
        dialect: Dialect,
    ) -> Result<()> {
        self.call(id, |reply| Command::Replace { conn, dialect, reply }).await
    }

    /// 关闭会话。有未提交的修改时按 `commit` 提交或回滚。
    pub async fn close(&self, id: SessionId, commit: bool) -> Result<()> {
        let result = self.call(id, |reply| Command::Close { commit, reply }).await;
        self.sessions.lock().map_err(|_| Error::internal("会话表锁已损坏"))?.remove(&id);
        result
    }
}

struct Actor {
    /// 断开后仍保留（关闭时尝试回滚）；`Replace` 换成新连接。
    conn: Option<Box<dyn Connection>>,
    ctx: ExecContext,
    policy: Arc<dyn Policy>,
    metadata: Arc<MetadataCache>,
    info: Arc<Mutex<SessionInfo>>,
    cancel: Arc<Mutex<CancelHandle>>,
    rx: mpsc::Receiver<Command>,
}

fn reply<T>(tx: oneshot::Sender<Result<T>>, value: Result<T>) {
    // 调用方已经不等结果（如请求被中断）时丢弃
    let _ = tx.send(value);
}

fn is_disconnect(e: &Error) -> bool {
    matches!(e, Error::Connect { .. })
}

impl Actor {
    fn update(&self, f: impl FnOnce(&mut SessionInfo)) {
        if let Ok(mut i) = self.info.lock() {
            f(&mut i);
        }
    }

    fn snapshot(&self) -> SessionInfo {
        self.info.lock().map(|i| i.clone()).unwrap_or_else(|p| p.into_inner().clone())
    }

    fn note_error<T>(&self, r: &Result<T>) {
        if let Err(e) = r
            && is_disconnect(e)
        {
            self.update(|i| i.broken = true);
        }
    }

    /// 空闲时的命令循环。
    async fn run(mut self) {
        while let Some(cmd) = self.rx.recv().await {
            match cmd {
                Command::Execute { req, reply: tx } => {
                    if self.snapshot().broken || self.conn.is_none() {
                        reply(tx, Err(Error::connect("连接已断开，请先重新连接")));
                        continue;
                    }
                    let steps = match plan(&self.ctx.dialect, &req) {
                        Ok(s) => s,
                        Err(e) => {
                            reply(tx, Err(e));
                            continue;
                        }
                    };
                    reply(tx, Ok(()));
                    self.update(|i| i.busy = true);
                    let closed = self.run_script(steps, &req).await;
                    self.update(|i| i.busy = false);
                    if let Some((commit, tx)) = closed {
                        reply(tx, self.close(commit).await);
                        return;
                    }
                }
                Command::Next { reply: tx, .. } => reply(tx, Ok(None)),
                Command::Schema { set, reply: tx } => {
                    let r = self.schema(set.as_deref()).await;
                    self.note_error(&r);
                    reply(tx, r);
                }
                Command::SetAutocommit { on, reply: tx } => {
                    let r = match self.conn.as_mut() {
                        Some(c) => c.set_autocommit(on).await,
                        None => Err(Error::connect("连接已断开")),
                    };
                    if r.is_ok() {
                        // 切回自动提交时驱动会提交未提交的修改
                        self.update(|i| {
                            i.autocommit = on;
                            if on {
                                i.in_transaction = false;
                            }
                        });
                    }
                    self.note_error(&r);
                    reply(tx, r);
                }
                Command::Commit { reply: tx } => {
                    let r = match self.conn.as_mut() {
                        Some(c) => c.commit().await,
                        None => Err(Error::connect("连接已断开")),
                    };
                    if r.is_ok() {
                        self.update(|i| i.in_transaction = false);
                    }
                    self.note_error(&r);
                    reply(tx, r);
                }
                Command::Rollback { reply: tx } => {
                    let r = match self.conn.as_mut() {
                        Some(c) => c.rollback().await,
                        None => Err(Error::connect("连接已断开")),
                    };
                    if r.is_ok() {
                        self.update(|i| i.in_transaction = false);
                    }
                    self.note_error(&r);
                    reply(tx, r);
                }
                Command::Replace { conn, dialect, reply: tx } => {
                    if let Some(old) = self.conn.take() {
                        let _ = old.close().await;
                    }
                    if let Ok(mut c) = self.cancel.lock() {
                        *c = conn.cancel_handle();
                    }
                    self.conn = Some(conn);
                    self.ctx.dialect = dialect;
                    self.update(|i| {
                        i.broken = false;
                        i.autocommit = true;
                        i.in_transaction = false;
                        i.variant = dialect.variant();
                    });
                    // 新连接回到登录时的默认 schema
                    let _ = self.schema(None).await;
                    reply(tx, Ok(()));
                }
                Command::Close { commit, reply: tx } => {
                    reply(tx, self.close(commit).await);
                    return;
                }
            }
        }
        // 管理器被丢弃：尽量回滚并断开
        let _ = self.close(false).await;
    }

    /// 切换（`set` 为 Some 时）并重新读取当前 schema。
    async fn schema(&mut self, set: Option<&str>) -> Result<()> {
        let Some(conn) = self.conn.as_mut() else { return Err(Error::connect("连接已断开")) };
        if let Some(name) = set {
            run_quiet(conn, &self.ctx.dialect.set_schema_sql(name)).await?;
        }
        let schema = query_schema(conn, &self.ctx.dialect).await?;
        self.update(|i| i.schema = schema);
        Ok(())
    }

    /// 关闭连接；有未提交的修改时按 `commit` 提交或回滚。
    async fn close(&mut self, commit: bool) -> Result<()> {
        let Some(mut conn) = self.conn.take() else { return Ok(()) };
        let state = self.snapshot();
        let mut result = Ok(());
        if state.in_transaction && !state.broken {
            result = if commit { conn.commit().await } else { conn.rollback().await };
        }
        let closed = conn.close().await;
        result.and(closed)
    }
}

/// 执行期间等来的请求。
enum Req {
    Next(usize, oneshot::Sender<Result<Option<SessionEvent>>>),
    Close(bool, oneshot::Sender<Result<()>>),
}

fn busy() -> Error {
    Error::Busy { message: "正在执行，请等待完成或先取消".into() }
}

/// 执行期间处理命令：`Next` 时队列里有事件就直接交出，队列空了才把请求交给执行循环去取数据；
/// 其他命令一律报 `Busy`。管理器被丢弃时按回滚关闭。
async fn take_request(rx: &mut mpsc::Receiver<Command>, queue: &mut VecDeque<SessionEvent>) -> Req {
    loop {
        let Some(cmd) = rx.recv().await else {
            let (tx, _) = oneshot::channel();
            return Req::Close(false, tx);
        };
        match cmd {
            Command::Next { max_rows, reply: tx } => match queue.pop_front() {
                Some(ev) => reply(tx, Ok(Some(ev))),
                None => return Req::Next(max_rows, tx),
            },
            Command::Close { commit, reply: tx } => return Req::Close(commit, tx),
            Command::Execute { reply: tx, .. } => reply(tx, Err(busy())),
            Command::SetAutocommit { reply: tx, .. }
            | Command::Schema { reply: tx, .. }
            | Command::Commit { reply: tx }
            | Command::Rollback { reply: tx }
            | Command::Replace { reply: tx, .. } => reply(tx, Err(busy())),
        }
    }
}

fn convert(ev: ExecEvent, affected: &mut Option<u64>) -> SessionEvent {
    match ev {
        ExecEvent::ResultStart { columns } => SessionEvent::ResultStart { columns },
        ExecEvent::Rows(rows) => SessionEvent::Rows { rows },
        ExecEvent::ResultEnd => SessionEvent::ResultEnd,
        ExecEvent::Affected(count) => {
            *affected = Some(affected.unwrap_or(0) + count);
            SessionEvent::Affected { count }
        }
        ExecEvent::Message(text) => SessionEvent::Message { text },
    }
}

impl Actor {
    /// 依次执行选出的语句，按 `Next` 请求逐个交出事件。执行中途会话被关闭时返回关闭请求。
    async fn run_script(
        &mut self,
        steps: Vec<Step>,
        req: &ExecRequest,
    ) -> Option<(bool, oneshot::Sender<Result<()>>)> {
        let Actor { conn, ctx, policy, metadata, info, rx, .. } = self;
        let conn = conn.as_mut()?;
        let set = |f: &dyn Fn(&mut SessionInfo)| {
            if let Ok(mut i) = info.lock() {
                f(&mut i);
            }
        };
        let mut queue: VecDeque<SessionEvent> = VecDeque::new();
        // 一个等着数据的 Next 请求：队列空时由执行循环取数据后回复
        let mut pending: Option<(usize, oneshot::Sender<Result<Option<SessionEvent>>>)> = None;

        let total = steps.len();
        'script: for step in steps {
            let index = step.index;
            // 执行计划按生成的语句（EXPLAIN …）分类：它不修改数据，不需要确认
            let stmt = classify(&ctx.dialect, &step.texts[0]);
            let start = SessionEvent::StatementStart {
                index,
                total,
                start: step.start,
                end: step.end,
                info: stmt.clone(),
            };
            match policy.before_execute(ctx, &stmt).await {
                Decision::Allow => {}
                // 确认只对第一条（就是刚确认的那条）有效
                Decision::Confirm { .. } if req.confirmed && index == 0 => {}
                Decision::Confirm { reason } => {
                    queue.push_back(SessionEvent::ConfirmRequired {
                        index,
                        start: step.start,
                        end: step.end,
                        reason,
                    });
                    break 'script;
                }
                Decision::Deny { reason } => {
                    // 先发开始事件，调用方才知道被拒绝的是哪一段
                    queue.push_back(start);
                    queue.push_back(SessionEvent::Failed {
                        index,
                        error: Error::Denied { reason },
                        offset: None,
                    });
                    if req.continue_on_error {
                        continue 'script;
                    }
                    break 'script;
                }
            }
            queue.push_back(start);
            // 惰性执行：调用方取走前面的事件、再来要数据时才真正开始执行。这样 StatementStart 能先送到，
            // 调用方在语句执行期间随时可以取消
            loop {
                let (m, tx) = match pending.take() {
                    Some(p) => p,
                    None => match take_request(rx, &mut queue).await {
                        Req::Next(m, tx) => (m, tx),
                        Req::Close(commit, tx) => return Some((commit, tx)),
                    },
                };
                match queue.pop_front() {
                    Some(ev) => reply(tx, Ok(Some(ev))),
                    None => {
                        pending = Some((m, tx));
                        break;
                    }
                }
            }
            let started = Instant::now();
            let mut affected = None;
            let mut result: Result<()> = Ok(());
            for text in &step.texts {
                result = match conn.execute(text, &[]).await {
                    Err(e) => Err(e),
                    Ok(mut stream) => loop {
                        let (max_rows, tx) = match pending.take() {
                            Some((m, tx)) => match queue.pop_front() {
                                Some(ev) => {
                                    reply(tx, Ok(Some(ev)));
                                    continue;
                                }
                                None => (m, tx),
                            },
                            None => match take_request(rx, &mut queue).await {
                                Req::Next(m, tx) => (m, tx),
                                Req::Close(commit, tx) => return Some((commit, tx)),
                            },
                        };
                        match stream.next(max_rows).await {
                            Ok(Some(ev)) => reply(tx, Ok(Some(convert(ev, &mut affected)))),
                            Ok(None) => {
                                pending = Some((max_rows, tx));
                                break Ok(());
                            }
                            Err(e) => {
                                pending = Some((max_rows, tx));
                                break Err(e);
                            }
                        }
                    },
                };
                if result.is_err() {
                    break;
                }
            }
            let outcome = match &result {
                Ok(()) => Outcome::Succeeded { affected },
                Err(Error::Cancelled) => Outcome::Cancelled,
                Err(e) => Outcome::Failed { message: e.to_string() },
            };
            policy.after_execute(ctx, &stmt, &outcome).await;
            match result {
                Ok(()) => {
                    queue.push_back(SessionEvent::StatementEnd {
                        index,
                        elapsed_ms: started.elapsed().as_millis() as u64,
                    });
                    if stmt.kind == StatementKind::Ddl {
                        metadata.invalidate_objects(&ctx.profile_id, &stmt.objects);
                    }
                    let autocommit = info.lock().map(|i| i.autocommit).unwrap_or(true);
                    match (stmt.kind, stmt.keyword.as_str()) {
                        (StatementKind::Dml, _) if !autocommit => set(&|i| i.in_transaction = true),
                        (StatementKind::Tcl, "COMMIT" | "ROLLBACK") => set(&|i| i.in_transaction = false),
                        _ => {}
                    }
                    if may_change_schema(&stmt.keyword)
                        && let Ok(schema) = query_schema(conn, &ctx.dialect).await
                    {
                        set(&|i| i.schema = schema.clone());
                    }
                }
                Err(error) => {
                    let disconnected = is_disconnect(&error);
                    if disconnected {
                        set(&|i| i.broken = true);
                    }
                    let stop = disconnected || matches!(error, Error::Cancelled) || !req.continue_on_error;
                    let offset = error_offset(&error, &step);
                    queue.push_back(SessionEvent::Failed { index, error, offset });
                    if stop {
                        break 'script;
                    }
                }
            }
        }
        queue.push_back(SessionEvent::Done);

        // 交出剩下的事件，最后一个是 Done
        if let Some((_, tx)) = pending.take() {
            reply(tx, Ok(queue.pop_front()));
        }
        // 最后一个事件交出后立即返回，之后的命令（提交、回滚等）由空闲循环处理
        while !queue.is_empty() {
            let cmd = rx.recv().await?;
            match cmd {
                Command::Next { reply: tx, .. } => reply(tx, Ok(queue.pop_front())),
                Command::Close { commit, reply: tx } => return Some((commit, tx)),
                Command::Execute { reply: tx, .. } => reply(tx, Err(busy())),
                Command::SetAutocommit { reply: tx, .. }
                | Command::Schema { reply: tx, .. }
                | Command::Commit { reply: tx }
                | Command::Rollback { reply: tx }
                | Command::Replace { reply: tx, .. } => reply(tx, Err(busy())),
            }
        }
        None
    }
}

    /// 取完全部事件（每批最多 2 行）。
    async fn drain(m: &SessionManager, id: SessionId) -> Vec<SessionEvent> {
        let mut out = Vec::new();
        while let Some(ev) = m.next(id, 2).await.unwrap() {
            let done = ev == SessionEvent::Done;
            out.push(ev);
            if done {
                break;
            }
        }
        out
    }

    fn kinds(events: &[SessionEvent]) -> Vec<String> {
        events
            .iter()
            .map(|e| match e {
                SessionEvent::StatementStart { index, .. } => format!("start{index}"),
                SessionEvent::ResultStart { .. } => "result".into(),
                SessionEvent::Rows { rows } => format!("rows{}", rows.len()),
                SessionEvent::ResultEnd => "end".into(),
                SessionEvent::Affected { count } => format!("affected{count}"),
                SessionEvent::Message { .. } => "message".into(),
                SessionEvent::StatementEnd { index, .. } => format!("done{index}"),
                SessionEvent::ConfirmRequired { index, .. } => format!("confirm{index}"),
                SessionEvent::Failed { index, .. } => format!("failed{index}"),
                SessionEvent::Done => "DONE".into(),
            })
            .collect()
    }
```

```rust
//! 会话的 WebSocket 执行协议：`GET /api/sessions/{id}/ws`。
//!
//! 客户端发送（JSON，按 `type` 区分）：
//! - `{"type":"exec","sql":"…","confirmed":false,"max_rows":500}`：执行一段 SQL（可以多条语句）。`confirmed` 只对
//!   第一条语句有效（确认后从那条重新提交），后面需要确认的语句仍会停下。可选字段：
//!   - `continue_on_error`：某条语句失败后继续执行后面的；
//!   - `cursor`：只执行这个 UTF-16 偏移所在的那条语句；
//!   - `explain`：只生成执行计划（只能有一条语句，或用 `cursor` 选出一条）；
//! - `{"type":"fetch_more","max_rows":500}`：取下一批行；
//! - `{"type":"cancel"}`：取消正在执行的语句。
//!
//! 服务端发送 `SessionEvent`（`statement_start`、`result_start`、`rows`、`result_end`、`affected`、`message`、
//! `statement_end`、`confirm_required`、`failed`、`done`），协议层面的错误发 `{"type":"error","error":{…}}`。
//! 每发出一个装满 `max_rows` 行的 `rows` 事件就停下，等客户端 `fetch_more` 后再取下一批（背压）；
//! 不满一批说明结果集已取完，接着发后面的事件。其他事件不受限制。
//! 事件中的 `start`/`end`、`failed` 的 `offset` 是 UTF-16 代码单元偏移（与 Monaco 编辑器一致），不是字节偏移。
//! 连接断开时，正在执行的语句会被取消，剩余事件被丢弃，会话恢复空闲。

use std::sync::Arc;

use axum::extract::ws::{Message, WebSocket, WebSocketUpgrade};
use axum::extract::{Path, State};
use axum::response::Response;
use dblens_core::Error;
use dblens_service::{ExecRequest, Service, SessionEvent, SessionId};
use futures::{SinkExt, StreamExt};
use serde::Deserialize;
use tokio::sync::mpsc;

use crate::AppState;
use crate::error::ApiResult;

/// 未指定时每批最多的行数。
pub const DEFAULT_MAX_ROWS: usize = 500;

fn default_rows() -> usize {
    DEFAULT_MAX_ROWS
}

#[derive(Debug, Deserialize)]
#[serde(tag = "type", rename_all = "snake_case")]
enum ClientMsg {
    Exec {
        sql: String,
        #[serde(default)]
        confirmed: bool,
        #[serde(default = "default_rows")]
        max_rows: usize,
        #[serde(default)]
        continue_on_error: bool,
        /// UTF-16 偏移。
        #[serde(default)]
        cursor: Option<usize>,
        #[serde(default)]
        explain: bool,
    },
    FetchMore {
        #[serde(default = "default_rows")]
        max_rows: usize,
    },
    Cancel,
}

/// 字节偏移换算成 UTF-16 偏移。
pub fn utf16_offset(text: &str, byte: usize) -> usize {
    text.get(..byte.min(text.len())).map_or(0, |s| s.encode_utf16().count())
}

/// UTF-16 偏移换算成字节偏移；落在字符中间时取该字符的起点，超出末尾时取文本长度。
pub fn byte_offset(text: &str, utf16: usize) -> usize {
    let mut units = 0;
    for (byte, c) in text.char_indices() {
        units += c.len_utf16();
        if units > utf16 {
            return byte;
        }
    }
    text.len()
}

/// 把事件中的字节偏移换成 UTF-16 偏移后序列化。
fn encode(event: SessionEvent, sql: &str) -> String {
    let event = match event {
        SessionEvent::StatementStart { index, total, start, end, info } => SessionEvent::StatementStart {
            index,
            total,
            start: utf16_offset(sql, start),
            end: utf16_offset(sql, end),
            info,
        },
        SessionEvent::ConfirmRequired { index, start, end, reason } => SessionEvent::ConfirmRequired {
            index,
            start: utf16_offset(sql, start),
            end: utf16_offset(sql, end),
            reason,
        },
        SessionEvent::Failed { index, error, offset } => {
            SessionEvent::Failed { index, error, offset: offset.map(|o| utf16_offset(sql, o)) }
        }
        other => other,
    };
    serde_json::to_string(&event).unwrap_or_else(|e| error_message(&Error::internal(e.to_string())))
}

fn error_message(e: &Error) -> String {
    serde_json::json!({"type": "error", "error": e}).to_string()
}

pub async fn upgrade(
    State(s): State<AppState>,
    Path(id): Path<SessionId>,
    ws: WebSocketUpgrade,
) -> ApiResult<Response> {
    s.service.sessions.info(id)?;
    let service = s.service.clone();
    Ok(ws.on_upgrade(move |socket| run(socket, service, id)))

/// 从会话取事件发给客户端；每发出装满的一批行就等一次 `fetch_more`。
async fn pump(
    service: Arc<Service>,
    id: SessionId,
    sql: String,
    mut max_rows: usize,
    out: mpsc::Sender<String>,
    mut credit: mpsc::Receiver<usize>,
) {
    loop {
        match service.sessions.next(id, max_rows).await {
            Ok(Some(event)) => {
                // 装满一批才等 fetch_more：不满说明结果集已经取完，接着发后面的事件
                let full = matches!(&event, SessionEvent::Rows { rows } if rows.len() >= max_rows);
                let done = event == SessionEvent::Done;
                let _ = out.send(encode(event, &sql)).await;
                if done {
                    return;
                }
                if full {
                    match credit.recv().await {
                        Some(n) => max_rows = n.max(1),
                        None => break,
                    }
                }
            }
            Ok(None) => return,
            Err(e) => {
                let _ = out.send(error_message(&e)).await;
                return;
            }
        }
    }
    // 客户端已断开：取消并取完剩余事件，让会话回到空闲
    let _ = service.sessions.cancel(id).await;
    while let Ok(Some(event)) = service.sessions.next(id, 1000).await {
        if event == SessionEvent::Done {
            break;
        }
    }
}

async fn run(socket: WebSocket, service: Arc<Service>, id: SessionId) {
    let (mut sink, mut stream) = socket.split();
    let (out, mut outbox) = mpsc::channel::<String>(64);
    let writer = tokio::spawn(async move {
        while let Some(text) = outbox.recv().await {
            if sink.send(Message::Text(text.into())).await.is_err() {
                break;
            }
        }
    });
    let mut credit: Option<mpsc::Sender<usize>> = None;
    let mut running: Option<tokio::task::JoinHandle<()>> = None;

    while let Some(Ok(msg)) = stream.next().await {
        let text = match msg {
            Message::Text(t) => t,
            Message::Close(_) => break,
            _ => continue,
        };
        let msg: ClientMsg = match serde_json::from_str(&text) {
            Ok(m) => m,
            Err(e) => {
                let _ = out.send(error_message(&Error::invalid(format!("无法解析的消息：{e}")))).await;
                continue;
            }
        };
        match msg {
            ClientMsg::Exec { sql, confirmed, max_rows, continue_on_error, cursor, explain } => {
                if running.as_ref().is_some_and(|h| !h.is_finished()) {
                    let busy = Error::Busy { message: "上一段 SQL 还没有执行完".into() };
                    let _ = out.send(error_message(&busy)).await;
                    continue;
                }
                let cursor = cursor.map(|c| byte_offset(&sql, c));
                let req = ExecRequest { sql: sql.clone(), confirmed, continue_on_error, cursor, explain };
                if let Err(e) = service.sessions.submit(id, req).await {
                    let _ = out.send(error_message(&e)).await;
                    continue;
                }
                let (tx, rx) = mpsc::channel(1);
                credit = Some(tx);
                running =
                    Some(tokio::spawn(pump(service.clone(), id, sql, max_rows.max(1), out.clone(), rx)));
            }
            ClientMsg::FetchMore { max_rows } => {
                if let Some(c) = &credit {
                    let _ = c.try_send(max_rows);
                }
            }
            ClientMsg::Cancel => {
                if let Err(e) = service.sessions.cancel(id).await {
                    let _ = out.send(error_message(&e)).await;
                }
            }
        }
    }

    // 连接断开：还在执行就取消（pump 会取完剩余事件）
    if running.as_ref().is_some_and(|h| !h.is_finished()) {
        let _ = service.sessions.cancel(id).await;
    }
    drop(credit);
    if let Some(h) = running {
        let _ = h.await;
    }
    drop(out);
    let _ = writer.await;
}
```

```ts
// 会话 WebSocket 执行协议（docs/notes/m0-4b-api.md、docs/notes/m1b1-editor.md）：发送 exec / fetch_more / cancel，
// 把服务端事件归并成界面要显示的状态。归并是纯函数，单独测试；ExecSocket 只负责收发。

import type { ErrorBody } from './api'

/** 单元格值：{"t": 类型, "v": 值}，Null 没有 v。 */
export interface CellValue {
  t: string
  v?: unknown
}

export interface ColumnMeta {
  name: string
  type_name: string
  nullable: boolean | null
}

export type ServerEvent =
  | { type: 'statement_start'; index: number; total: number; start: number; end: number; info: { keyword: string } }
  | { type: 'result_start'; columns: ColumnMeta[] }
  | { type: 'rows'; rows: CellValue[][] }
  | { type: 'result_end' }
  | { type: 'affected'; count: number }
  | { type: 'message'; text: string }
  | { type: 'statement_end'; index: number; elapsed_ms: number }
  | { type: 'confirm_required'; index: number; start: number; end: number; reason: string }
  | { type: 'failed'; index: number; error: ErrorBody; offset?: number }
  | { type: 'done' }
  | { type: 'error'; error: ErrorBody }

export interface ResultSet {
  columns: ColumnMeta[]
  rows: CellValue[][]
  /** 结果集已取完。 */
  complete: boolean
}

export interface StatementState {
  index: number
  /** 是这次提交的最后一条语句。前面语句的结果要取完，后面的语句才会执行。 */
  last: boolean
  /** 语句在提交的 SQL 中的 UTF-16 偏移，可直接用于 String.prototype.slice。 */
  start: number
  end: number
  keyword: string
  results: ResultSet[]
  affected: number | null
  messages: string[]
  elapsedMs: number | null
  error: ErrorBody | null
  /** 数据库报告的出错位置（UTF-16 偏移）；没有时为 null。 */
  errorOffset: number | null
}

export interface ExecState {
  statements: StatementState[]
  /** 一批装满了 max_rows 行，服务端在等 fetch_more。 */
  waitingForMore: boolean
  done: boolean
  /** 第 `index` 条语句需要确认，脚本停在这里（这条还没有开始）。`start`/`end` 是它的 UTF-16 范围。 */
  confirm: { index: number; start: number; end: number; reason: string } | null
  /** 协议层面的错误（消息无法解析、上一段还没执行完等）。 */
  protocolError: ErrorBody | null
}

export const initialExec: ExecState = {
  statements: [],
  waitingForMore: false,
  done: false,
  confirm: null,
  protocolError: null,
}

function updateLast(state: ExecState, f: (s: StatementState) => StatementState): ExecState {
  const last = state.statements.length - 1
  if (last < 0) return state
  return { ...state, statements: state.statements.map((s, i) => (i === last ? f(s) : s)) }
}

function updateLastResult(s: StatementState, f: (r: ResultSet) => ResultSet): StatementState {
  const last = s.results.length - 1
  if (last < 0) return s
  return { ...s, results: s.results.map((r, i) => (i === last ? f(r) : r)) }
}

/** 把一个服务端事件归并进状态。`maxRows` 是本次请求的每批行数，用来判断服务端是否在等 fetch_more。 */
export function reduceExec(state: ExecState, ev: ServerEvent, maxRows: number): ExecState {
  switch (ev.type) {
    case 'statement_start':
      return {
        ...state,
        statements: [
          ...state.statements,
          {
            index: ev.index,
            last: ev.index + 1 >= ev.total,
            start: ev.start,
            end: ev.end,
            keyword: ev.info.keyword,
            results: [],
            affected: null,
            messages: [],
            elapsedMs: null,
            error: null,
            errorOffset: null,
          },
        ],
      }
    case 'result_start':
      return updateLast(state, (s) => ({ ...s, results: [...s.results, { columns: ev.columns, rows: [], complete: false }] }))
    case 'rows':
      return {
        ...updateLast(state, (s) => updateLastResult(s, (r) => ({ ...r, rows: [...r.rows, ...ev.rows] }))),
        waitingForMore: ev.rows.length >= maxRows,
      }
    case 'result_end':
      return { ...updateLast(state, (s) => updateLastResult(s, (r) => ({ ...r, complete: true }))), waitingForMore: false }
    case 'affected':
      return updateLast(state, (s) => ({ ...s, affected: ev.count }))
    case 'message':
      return updateLast(state, (s) => ({ ...s, messages: [...s.messages, ev.text] }))
    case 'statement_end':
      return updateLast(state, (s) => ({ ...s, elapsedMs: ev.elapsed_ms }))
    case 'confirm_required':
      return { ...state, confirm: { index: ev.index, start: ev.start, end: ev.end, reason: ev.reason } }
    case 'failed':
      return {
        ...updateLast(state, (s) => ({ ...s, error: ev.error, errorOffset: ev.offset ?? null })),
        waitingForMore: false,
      }
    case 'done':
      return { ...state, done: true, waitingForMore: false }
    case 'error':
      return { ...state, protocolError: ev.error, waitingForMore: false }
  }
}

export const DEFAULT_MAX_ROWS = 500

export interface ExecOptions {
  /** 已确认：需要确认的语句直接执行。 */
  confirmed?: boolean
  /** 某条语句失败后继续执行后面的。 */
  continueOnError?: boolean
  /** 只执行这个 UTF-16 偏移所在的那条语句。 */
  cursor?: number
  /** 只生成执行计划。 */
  explain?: boolean
  maxRows?: number
}

/** 一个会话的执行连接。onEvent 收到服务端的每个事件；连接断开时服务端会取消正在执行的语句。 */
export class ExecSocket {
  private ws: WebSocket
  private ready: Promise<void>

  constructor(url: string, onEvent: (ev: ServerEvent) => void, onClose: () => void) {
    this.ws = new WebSocket(url)
    this.ready = new Promise((resolve, reject) => {
      this.ws.addEventListener('open', () => resolve(), { once: true })
      this.ws.addEventListener('error', () => reject(new Error('WebSocket 连接失败')), { once: true })
    })
    this.ws.addEventListener('message', (m) => onEvent(JSON.parse(m.data as string) as ServerEvent))
    this.ws.addEventListener('close', onClose)
  }

  private async send(msg: object) {
    await this.ready
    this.ws.send(JSON.stringify(msg))
  }

  exec(sql: string, o: ExecOptions = {}) {
    return this.send({
      type: 'exec',
      sql,
      confirmed: o.confirmed ?? false,
      continue_on_error: o.continueOnError ?? false,
      cursor: o.cursor ?? null,
      explain: o.explain ?? false,
      max_rows: o.maxRows ?? DEFAULT_MAX_ROWS,
    })
  }

  fetchMore(maxRows = DEFAULT_MAX_ROWS) {
    return this.send({ type: 'fetch_more', max_rows: maxRows })
  }

  cancel() {
    return this.send({ type: 'cancel' })
  }

  close() {
    this.ws.close()
  }
}

/** 单元格的显示文本。 */
export function cellText(c: CellValue): string {
  if (c.t === 'null') return 'NULL'
  if (c.t === 'lob') {
    const lob = c.v as { preview: string; total_len: number; truncated: boolean; binary: boolean }
    return lob.truncated ? `${lob.binary ? '(二进制) ' : ''}${lob.preview}…（共 ${lob.total_len}）` : lob.preview
  }
  if (c.t === 'other') return (c.v as { text: string }).text
  if (typeof c.v === 'object') return JSON.stringify(c.v)
  return String(c.v)
}
```

执行器的要点：
- **偏移**：执行选中部分、确认后续跑时，提交的只是全文的一段。收到事件后加上这段的起点（`shiftEvent`），状态里存的都是全文中的偏移。
- **自动续取**：服务端在等 `fetch_more` 时：
  - 不是最后一条语句：自动续取到底（取完后面的语句才能执行）；
  - 最后一条：等表格滚动到底部。
  - 准备阶段发现，"执行全部"时第一条的结果不滚到底，后面的语句就一直不执行。
- **行数上限**：达到上限时取消这条语句，并记下 `capped`。
- **确认**：
  - 确认后：执行单条语句时原样重发，否则从需要确认的那条截取后重发（`resumeAfterConfirm`），都带 `confirmed`；
  - 取消确认：只记下 `declined`，确认框关掉，消息中说明。
- **会话**：第一次执行时打开（密码库锁定时先解锁）。服务重启过、会话不存在时重新打开。

```ts
// 查询标签页（M1 设计文档 1 节）：每个标签绑定一个连接，有自己的 SQL。打开的标签和其中的 SQL 存进设置
// （ui.tabs），下次启动时恢复；会话不恢复，第一次执行时再打开。

import { create } from 'zustand'

import { useSettings } from '../../shared/settings'

export const TABS_KEY = 'ui.tabs'

export interface QueryTab {
  id: string
  /** 「查询1」这样的标题。 */
  title: string
  profileId: string
  sql: string
}

interface Persisted {
  tabs: QueryTab[]
  active: string | null
  /** 下一个新标签的编号。 */
  seq: number
}

/** 解析保存的标签；格式不对时丢弃（界面设置不值得报错）。 */
export function parseTabs(raw: string | undefined): Persisted {
  const empty: Persisted = { tabs: [], active: null, seq: 1 }
  if (!raw) return empty
  try {
    const v = JSON.parse(raw) as Partial<Persisted>
    const tabs = (Array.isArray(v.tabs) ? v.tabs : []).filter(
      (t): t is QueryTab =>
        typeof t?.id === 'string' &&
        typeof t.title === 'string' &&
        typeof t.profileId === 'string' &&
        typeof t.sql === 'string',
    )
    const active = tabs.some((t) => t.id === v.active) ? (v.active as string) : (tabs[0]?.id ?? null)
    return { tabs, active, seq: typeof v.seq === 'number' && v.seq > 0 ? v.seq : tabs.length + 1 }
  } catch {
    return empty
  }
}

interface TabsState extends Persisted {
  /** 新建标签并切换过去，返回它的 id。 */
  open(profileId: string, sql?: string): string
  close(id: string): void
  activate(id: string): void
  setSql(id: string, sql: string): void
  setProfile(id: string, profileId: string): void
  restore(raw: string | undefined): void
}

let ids = 0
const newId = () => `t${Date.now().toString(36)}${(ids++).toString(36)}`

export const useTabs = create<TabsState>()((set, get) => ({
  tabs: [],
  active: null,
  seq: 1,
  open: (profileId, sql = '') => {
    const { tabs, seq } = get()
    const tab = { id: newId(), title: `查询${seq}`, profileId, sql }
    set({ tabs: [...tabs, tab], active: tab.id, seq: seq + 1 })
    return tab.id
  },
  close: (id) => {
    const { tabs, active } = get()
    const i = tabs.findIndex((t) => t.id === id)
    if (i < 0) return
    const rest = tabs.filter((t) => t.id !== id)
    // 关闭当前标签时切到右边的，没有就左边的
    const next = active === id ? (rest[i] ?? rest[i - 1])?.id ?? null : active
    set({ tabs: rest, active: next })
  },
  activate: (active) => set({ active }),
  setSql: (id, sql) => set({ tabs: get().tabs.map((t) => (t.id === id ? { ...t, sql } : t)) }),
  setProfile: (id, profileId) => set({ tabs: get().tabs.map((t) => (t.id === id ? { ...t, profileId } : t)) }),
  restore: (raw) => set(parseTabs(raw)),
}))

let stopPersist: (() => void) | null = null

/** 标签有变化时延迟保存到设置（输入 SQL 时不必每个字都保存）。重复调用不会重复订阅。返回取消订阅的函数。 */
export function persistTabs(delayMs = 800): () => void {
  if (stopPersist) return stopPersist
  let timer: ReturnType<typeof setTimeout> | undefined
  const unsubscribe = useTabs.subscribe((s, prev) => {
    if (s.tabs === prev.tabs && s.active === prev.active) return
    clearTimeout(timer)
    timer = setTimeout(() => {
      const { tabs, active, seq } = useTabs.getState()
      useSettings.getState().set(TABS_KEY, JSON.stringify({ tabs, active, seq }))
    }, delayMs)
  })
  stopPersist = () => {
    clearTimeout(timer)
    unsubscribe()
    stopPersist = null
  }
  return stopPersist
}
```

`web/src/features/editor/runner.ts`：

```ts
// 标签页的执行器：每个标签一个，持有它的会话和 WebSocket，把服务端事件归并进 useRuntime。
// 会话在第一次执行时打开（密码库锁定时先解锁）；关闭标签时关闭会话。
//
// 偏移：编辑器给的是全文中的 UTF-16 偏移。执行选中部分或确认后续跑时提交的只是全文的一段，
// 服务端事件里的偏移相对于提交的那段，收到后加上这段的起点（base），状态里存的都是全文中的偏移。

import { create } from 'zustand'

import { type Api, ApiError, type SessionInfo } from '../../shared/api'
import {
  DEFAULT_MAX_ROWS,
  type ExecOptions,
  ExecSocket,
  type ExecState,
  type ServerEvent,
  initialExec,
  reduceExec,
} from '../../shared/exec'
import { useSettings } from '../../shared/settings'
import { withVault } from '../connections/vault'
import { useTabs } from './tabs'

/** 结果行数上限的设置项，默认 10 万行（M1 设计文档第 4 节）。 */
export const ROW_CAP_KEY = 'ui.exec.row-cap'
export const DEFAULT_ROW_CAP = 100_000

export function rowCap(): number {
  const v = Number(useSettings.getState().values[ROW_CAP_KEY])
  return Number.isInteger(v) && v >= DEFAULT_MAX_ROWS ? v : DEFAULT_ROW_CAP
}

/** 一次提交：提交的 SQL、它在编辑器全文中的起点、执行选项。 */
export interface Submission {
  sql: string
  base: number
  opts: ExecOptions
}

export interface TabRuntime {
  session: SessionInfo | null
  /** 打开会话、连接服务失败等不属于某条语句的错误。 */
  error: string | null
  exec: ExecState
  running: boolean
  /** 当前结果来自执行计划。 */
  explain: boolean
  submission: Submission | null
  /** 执行时编辑器的全文：错误标记、失败语句的显示都按它取文本（之后的编辑不影响）。 */
  source: string
  continueOnError: boolean
  /** 结果行数达到上限、已停止读取的语句（statements 的下标）。 */
  capped: number | null
  /** 用户在确认框中选了取消：需要确认的那条和后面的语句都没有执行。 */
  declined: boolean
}

export const emptyRuntime: TabRuntime = {
  session: null,
  error: null,
  exec: initialExec,
  running: false,
  explain: false,
  submission: null,
  source: '',
  continueOnError: false,
  capped: null,
  declined: false,
}

interface RuntimeState {
  tabs: Record<string, TabRuntime>
  patch(id: string, p: Partial<TabRuntime>): void
  drop(id: string): void
}

export const useRuntime = create<RuntimeState>()((set, get) => ({
  tabs: {},
  patch: (id, p) => set({ tabs: { ...get().tabs, [id]: { ...(get().tabs[id] ?? emptyRuntime), ...p } } }),
  drop: (id) => {
    const tabs = { ...get().tabs }
    delete tabs[id]
    set({ tabs })
  },
}))

export function useTabRuntime(id: string | null): TabRuntime {
  return useRuntime((s) => (id ? (s.tabs[id] ?? emptyRuntime) : emptyRuntime))
}

/** 把服务端事件中的偏移换成编辑器全文中的偏移。 */
export function shiftEvent(ev: ServerEvent, base: number): ServerEvent {
  if (base === 0) return ev
  switch (ev.type) {
    case 'statement_start':
      return { ...ev, start: ev.start + base, end: ev.end + base }
    case 'confirm_required':
      return { ...ev, start: ev.start + base, end: ev.end + base }
    case 'failed':
      return ev.offset === undefined ? ev : { ...ev, offset: ev.offset + base }
    default:
      return ev
  }
}

/** 确认后接着执行的提交：执行单条语句时原样重发，否则从需要确认的那条开始截取。 */
export function resumeAfterConfirm(sub: Submission, confirmStart: number): Submission {
  const opts = { ...sub.opts, confirmed: true }
  if (sub.opts.cursor !== undefined) return { ...sub, opts }
  return { sql: sub.sql.slice(confirmStart - sub.base), base: confirmStart, opts }
}

function message(e: unknown): string {
  return e instanceof Error ? e.message : String(e)
}

export class Runner {
  private socket: ExecSocket | null = null

  constructor(
    private api: Api,
    readonly tabId: string,
  ) {}

  private rt(): TabRuntime {
    return useRuntime.getState().tabs[this.tabId] ?? emptyRuntime
  }

  private patch(p: Partial<TabRuntime>) {
    useRuntime.getState().patch(this.tabId, p)
  }

  /** 会话和 WebSocket 都就绪后返回会话；服务重启过、会话已不存在时重新打开。 */
  private async ensureSession(): Promise<SessionInfo> {
    let session = this.rt().session
    if (session && this.socket) return session
    if (session) {
      session = await this.api.session(session.id).catch((e: unknown) => {
        if (e instanceof ApiError && e.body.kind === 'not_found') return null
        throw e
      })
    }
    if (!session) {
      const tab = useTabs.getState().tabs.find((t) => t.id === this.tabId)
      if (!tab) throw new Error('标签页已关闭')
      session = await withVault(this.api, () => this.api.openSession(tab.profileId))
    }
    this.patch({ session, error: null })
    this.socket = new ExecSocket(
      this.api.wsUrl(session.id),
      (ev) => this.onEvent(ev),
      () => this.onClose(),
    )
    return session
  }

  /** 执行 sql；base 是它在编辑器全文 source 中的起点。正在执行时忽略。 */
  async run(sql: string, base: number, opts: ExecOptions = {}, source = sql) {
    if (this.rt().running) return
    const submission: Submission = {
      sql,
      base,
      opts: { ...opts, continueOnError: this.rt().continueOnError, maxRows: DEFAULT_MAX_ROWS },
    }
    this.patch({
      exec: initialExec,
      running: true,
      explain: !!opts.explain,
      submission,
      source,
      capped: null,
      declined: false,
      error: null,
    })
    try {
      await this.ensureSession()
      await this.socket!.exec(submission.sql, submission.opts)
    } catch (e) {
      this.patch({ running: false, error: message(e) })
    }
  }

  /** 用户确认后，从需要确认的那条语句接着执行。 */
  async confirm() {
    const { exec, submission } = this.rt()
    if (!exec.confirm || !submission || !this.socket) return
    const next = resumeAfterConfirm(submission, exec.confirm.start)
    this.patch({ exec: { ...exec, confirm: null, done: false }, running: true, submission: next })
    await this.socket.exec(next.sql, next.opts).catch((e: unknown) => this.patch({ running: false, error: message(e) }))
  }

  /** 用户取消确认：脚本已经停下，关掉确认框，消息中说明没有执行。 */
  dismissConfirm() {
    this.patch({ declined: true })
  }

  private onEvent(raw: ServerEvent) {
    const { submission, exec, capped } = this.rt()
    const ev = shiftEvent(raw, submission?.base ?? 0)
    const next = reduceExec(exec, ev, submission?.opts.maxRows ?? DEFAULT_MAX_ROWS)
    const p: Partial<TabRuntime> = { exec: next }
    if (ev.type === 'done' || ev.type === 'error') {
      p.running = false
      void this.refreshSession()
    }
    if (next.waitingForMore && capped === null) {
      const last = next.statements.at(-1)
      if ((last?.results.at(-1)?.rows.length ?? 0) >= rowCap()) {
        // 达到行数上限：不再续取，取消这条语句（服务端在等 fetch_more，不取消的话会话一直忙）。
        // 后面还有语句的话也不再执行
        p.capped = next.statements.length - 1
        void this.socket?.cancel()
      } else if (last && !last.last) {
        // 不是最后一条：取完它的结果，后面的语句才能执行；最后一条等表格滚动到底部再续取
        p.exec = { ...next, waitingForMore: false }
        void this.socket?.fetchMore()
      }
    }
    this.patch(p)
  }

  private onClose() {
    this.socket = null
    if (this.rt().running) this.patch({ running: false, error: '与 DBLens 服务的连接已断开' })
  }

  /** 表格滚动到底部时取下一批。 */
  fetchMore() {
    const { exec, capped } = this.rt()
    if (!exec.waitingForMore || capped !== null || !this.socket) return
    this.patch({ exec: { ...exec, waitingForMore: false } })
    void this.socket.fetchMore()
  }

  cancel() {
    void this.socket?.cancel()
  }

  setContinueOnError(on: boolean) {
    this.patch({ continueOnError: on })
  }

  async refreshSession() {
    const s = this.rt().session
    if (!s) return
    const session = await this.api.session(s.id).catch(() => null)
    if (session) this.patch({ session })
  }

  /** 会话级操作：需要时先打开会话，完成后更新会话信息。出错时抛给调用方显示。 */
  private async sessionOp(f: (id: number) => Promise<SessionInfo>) {
    const s = await this.ensureSession()
    this.patch({ session: await f(s.id) })
  }

  setAutocommit = (on: boolean) => this.sessionOp((id) => this.api.setAutocommit(id, on))
  commit = () => this.sessionOp((id) => this.api.commit(id))
  rollback = () => this.sessionOp((id) => this.api.rollback(id))
  setSchema = (schema: string) => this.sessionOp((id) => this.api.setSchema(id, schema))
  reconnect = () => this.sessionOp((id) => this.api.reconnect(id))

  /** 关闭会话（有未提交的修改时按 commit 提交或回滚）并丢掉状态。 */
  async dispose(commit = false) {
    const s = this.rt().session
    this.socket?.close()
    this.socket = null
    useRuntime.getState().drop(this.tabId)
    if (s) await this.api.closeSession(s.id, commit).catch((e: unknown) => console.warn('关闭会话失败', e))
  }
}

const runners = new Map<string, Runner>()

export function runnerFor(api: Api, tabId: string): Runner {
  let r = runners.get(tabId)
  if (!r) {
    r = new Runner(api, tabId)
    runners.set(tabId, r)
  }
  return r
}

/** 关闭标签页：关闭会话、丢掉执行器。 */
export async function disposeRunner(tabId: string, commit = false) {
  const r = runners.get(tabId)
  runners.delete(tabId)
  if (r) await r.dispose(commit)
  else useRuntime.getState().drop(tabId)
}
```

```ts
// 结果面板的内容（M1 设计文档 1 节）：每个结果集一个标签，失败的语句单独一个标签，最后是"消息"。
// 纯函数，按执行状态生成，单独测试。

import { errorText } from '../../shared/api'
import type { ExecState, StatementState } from '../../shared/exec'

export type ResultTab =
  | { key: string; kind: 'result'; label: string; stmt: number; result: number; rows: number; capped: boolean }
  | { key: string; kind: 'error'; label: string; stmt: number }
  | { key: 'messages'; kind: 'messages'; label: string }

/** 因达到行数上限被取消的语句不算失败。 */
function failed(s: StatementState, i: number, capped: number | null): boolean {
  return s.error !== null && !(i === capped && s.error.kind === 'cancelled')
}

export function resultTabs(exec: ExecState, explain: boolean, capped: number | null): ResultTab[] {
  const tabs: ResultTab[] = []
  let n = 0
  exec.statements.forEach((s, i) => {
    s.results.forEach((r, j) => {
      n += 1
      tabs.push({
        key: `r${i}.${j}`,
        kind: 'result',
        label: explain ? '执行计划' : `结果 ${n}`,
        stmt: i,
        result: j,
        rows: r.rows.length,
        capped: i === capped,
      })
    })
    if (failed(s, i, capped)) {
      const label = s.error?.kind === 'cancelled' ? `语句 ${i + 1} 已取消` : `语句 ${i + 1} 失败`
      tabs.push({ key: `e${i}`, kind: 'error', label, stmt: i })
    }
  })
  tabs.push({ key: 'messages', kind: 'messages', label: '消息' })
  return tabs
}

export interface MessageLine {
  level: 'info' | 'warn' | 'error'
  text: string
}

const ms = (v: number | null) => (v === null ? '' : `，${v} ms`)

/** "消息"标签的内容：每条语句的结果摘要、服务器消息、错误，以及确认（`declined`：用户取消了）和协议错误。 */
export function messageLines(exec: ExecState, capped: number | null, declined = false): MessageLine[] {
  const out: MessageLine[] = []
  exec.statements.forEach((s, i) => {
    const tag = `[语句 ${i + 1}]`
    for (const text of s.messages) out.push({ level: 'info', text: `${tag} ${text}` })
    s.results.forEach((r) => out.push({ level: 'info', text: `${tag} ${r.rows.length} 行${ms(s.elapsedMs)}` }))
    if (i === capped) {
      const rows = s.results.at(-1)?.rows.length ?? 0
      const rest = s.last ? '' : '，后面的语句没有执行'
      out.push({ level: 'warn', text: `${tag} 已读取 ${rows} 行，达到行数上限，停止读取${rest}` })
    } else if (s.error) {
      out.push({ level: 'error', text: `${tag} ${errorText(s.error)}` })
    } else if (s.affected !== null && s.results.length === 0) {
      // 有结果集的语句（如 Oracle 的执行计划先写计划表）不再报影响行数
      out.push({ level: 'info', text: `${tag} 影响 ${s.affected} 行${ms(s.elapsedMs)}` })
    } else if (s.results.length === 0 && s.elapsedMs !== null) {
      out.push({ level: 'info', text: `${tag} 执行成功${ms(s.elapsedMs)}` })
    }
  })
  if (exec.confirm) {
    const tag = `[语句 ${exec.statements.length + 1}]`
    out.push(
      declined
        ? { level: 'warn', text: `${tag} 已取消执行（${exec.confirm.reason}），后面的语句也没有执行` }
        : { level: 'warn', text: `${tag} 需要确认：${exec.confirm.reason}` },
    )
  }
  if (exec.protocolError) out.push({ level: 'error', text: errorText(exec.protocolError) })
  return out
}

/** 状态栏上最近一次执行的摘要：最后一个结果集的行数，或最后一条语句的影响行数，加上耗时。 */
export function lastSummary(exec: ExecState): string | null {
  const s = exec.statements.at(-1)
  if (!s || s.elapsedMs === null) return null
  const r = s.results.at(-1)
  if (r) return `${r.rows.length} 行 · ${s.elapsedMs} ms`
  if (s.affected !== null) return `影响 ${s.affected} 行 · ${s.elapsedMs} ms`
  return `${s.elapsedMs} ms`
}

/** 表格滚动到离已加载的末尾不到 `margin` 行、且服务端在等时，续取下一批。 */
export function shouldFetchMore(lastVisibleRow: number, loaded: number, waiting: boolean, margin = 100): boolean {
  return waiting && lastVisibleRow >= loaded - margin
}
```

```ts
// 单元格查看（设计稿第 3 部分）：文本、JSON、十六进制三种显示。纯函数，单独测试。

import { type CellValue, cellText } from '../../shared/exec'

interface LobPreview {
  preview: string
  total_len: number
  truncated: boolean
  binary: boolean
}

/** 文本视图：字符串和大字段显示完整的预览内容，其他值同表格。 */
export function cellFullText(c: CellValue): string {
  if (c.t === 'lob') return (c.v as LobPreview).preview
  if (c.t === 'string') return String(c.v)
  return cellText(c)
}

/** 大字段的说明：总长度，以及是否只取了预览。 */
export function lobNote(c: CellValue): string | null {
  if (c.t !== 'lob') return null
  const v = c.v as LobPreview
  return `${formatBytes(v.total_len)}${v.truncated ? ' · 已截断预览' : ''}`
}

export function formatBytes(n: number): string {
  if (n < 1024) return `${n} B`
  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
  return `${(n / 1024 / 1024).toFixed(1)} MB`
}

/** JSON 视图：能解析成对象或数组时缩进显示，否则为 null（不提供这个视图）。 */
export function prettyJson(c: CellValue): string | null {
  if (c.t === 'json') return JSON.stringify(c.v, null, 2)
  const text = cellFullText(c).trim()
  if (!text.startsWith('{') && !text.startsWith('[')) return null
  try {
    return JSON.stringify(JSON.parse(text), null, 2)
  } catch {
    return null
  }
}

/** 单元格的字节：二进制值按 base64 解码，其他按 UTF-8 编码显示文本。 */
export function cellBytes(c: CellValue): Uint8Array {
  if (c.t === 'bytes' && typeof c.v === 'string') return Uint8Array.from(atob(c.v), (ch) => ch.charCodeAt(0))
  return new TextEncoder().encode(cellFullText(c))
}

/** 十六进制视图：每行 16 字节，偏移、十六进制、可打印字符。最多显示 `max` 字节。 */
export function hexDump(bytes: Uint8Array, max = 64 * 1024): string {
  const lines: string[] = []
  const n = Math.min(bytes.length, max)
  for (let off = 0; off < n; off += 16) {
    const chunk = Array.from(bytes.subarray(off, Math.min(off + 16, n)))
    const hex = chunk.map((b) => b.toString(16).padStart(2, '0')).join(' ')
    const ascii = chunk.map((b) => (b >= 0x20 && b < 0x7f ? String.fromCharCode(b) : '.')).join('')
    lines.push(`${off.toString(16).padStart(8, '0')}  ${hex.padEnd(47)}  |${ascii}|`)
  }
  if (bytes.length > max) lines.push(`…（共 ${bytes.length} 字节，只显示前 ${max} 字节）`)
  return lines.join('\n')
}

/** 数值类型右对齐。 */
export function isNumeric(c: CellValue): boolean {
  return c.t === 'int' || c.t === 'u_int' || c.t === 'float' || c.t === 'decimal'
}
```

```tsx
// 快捷键（M1 设计文档 1.3 节）注册为编辑器动作，焦点在编辑器里时由 Monaco 处理；错误标记来自执行结果。

import Editor, { type OnMount } from '@monaco-editor/react'
import { useEffect, useRef } from 'react'

import { useSetting } from '../../shared/settings'
import { THEME_KEY, parseThemePref, resolveTheme, useSystemDark } from '../../shared/theme'
import { type EditorCommand, sendCommand } from './commands'
import { useCursor } from './cursor'
import type { ErrorMark } from './markers'
import { monaco } from './monaco'

export type MonacoEditor = Parameters<OnMount>[0]

const { KeyMod, KeyCode } = monaco
const ACTIONS: { id: EditorCommand; label: string; keys: number[] }[] = [
  { id: 'run-current', label: '执行当前语句', keys: [KeyMod.CtrlCmd | KeyCode.Enter] },
  { id: 'run-all', label: '执行全部', keys: [KeyCode.F5] },
  { id: 'cancel', label: '停止', keys: [KeyMod.CtrlCmd | KeyCode.PauseBreak, KeyMod.CtrlCmd | KeyCode.Period] },
  { id: 'format', label: '格式化', keys: [KeyMod.Shift | KeyMod.Alt | KeyCode.KeyF] },
  { id: 'explain', label: '执行计划', keys: [] },
]

export interface SqlEditorProps {
  path: string
  language: string
  defaultValue: string
  marks: ErrorMark[]
  onChange(value: string): void
  onReady(editor: MonacoEditor | null): void
}

export default function SqlEditor({ path, language, defaultValue, marks, onChange, onReady }: SqlEditorProps) {
  const systemDark = useSystemDark()
  const theme = resolveTheme(parseThemePref(useSetting(THEME_KEY)), systemDark)
  const editorRef = useRef<MonacoEditor | null>(null)

  const onMount: OnMount = (editor) => {
    editorRef.current = editor
    for (const a of ACTIONS) {
      editor.addAction({
        id: `dblens.${a.id}`,
        label: a.label,
        keybindings: a.keys,
        contextMenuGroupId: a.id === 'cancel' ? undefined : '0_dblens',
        run: () => sendCommand(a.id),
      })
    }
    editor.onDidChangeCursorPosition((e) =>
      useCursor.setState({ line: e.position.lineNumber, column: e.position.column }),
    )
    onReady(editor)
    editor.focus()
  }
  useEffect(() => () => onReady(null), [onReady])
  // 切换或新建标签后直接可以输入
  useEffect(() => editorRef.current?.focus(), [path])

  // 错误标记：执行结果变化时整体替换
  useEffect(() => {
    const model = editorRef.current?.getModel()
    if (!model || model.uri.path !== monaco.Uri.parse(path).path) return
    monaco.editor.setModelMarkers(
      model,
      'dblens',
      marks.map((m) => {
        const s = model.getPositionAt(m.start)
        const e = model.getPositionAt(m.end)
        return {
          severity: monaco.MarkerSeverity.Error,
          message: m.message,
          startLineNumber: s.lineNumber,
          startColumn: s.column,
          endLineNumber: e.lineNumber,
          endColumn: e.column,
        }
      }),
    )
  }, [marks, path])

  return (
    <Editor
      path={path}
      // 切换标签时保留模型（撤销历史）；关闭标签时由工作区删除
      keepCurrentModel
      language={language}
      defaultValue={defaultValue}
      theme={theme === 'dark' ? 'dblens-dark' : 'dblens-light'}
      onMount={onMount}
```

```tsx
// 一个查询标签页的内容：工具栏、提示条、编辑器和结果面板（可拖动分隔线，位置存进设置）、确认框。
// 处理编辑器命令：执行当前语句（有选中时执行选中部分）、执行全部、停止、格式化、执行计划。

import { Alert, App, Button } from 'antd'
import { useCallback, useEffect, useMemo, useRef } from 'react'
import { Group, Panel, Separator, useDefaultLayout } from 'react-resizable-panels'

import { useApi } from '../../app/context'
import { type Profile, errorText } from '../../shared/api'
import { layoutStorage } from '../../shared/settings'
import ResultsPanel from '../results/ResultsPanel'
import ConfirmDialog from './ConfirmDialog'
import Toolbar from './Toolbar'
import { onCommand } from './commands'
import { editorLanguage, formatSql } from './dialect'
import { errorMarks } from './markers'
import { runnerFor, useTabRuntime } from './runner'
import SqlEditor, { type MonacoEditor } from './SqlEditor'
import { type QueryTab, useTabs } from './tabs'

export default function QueryView({ tab, profiles }: { tab: QueryTab; profiles: Profile[] }) {
  const api = useApi()
  const { message } = App.useApp()
  const runner = runnerFor(api, tab.id)
  const rt = useTabRuntime(tab.id)
  const profile = profiles.find((p) => p.id === tab.profileId)
  const editor = useRef<MonacoEditor | null>(null)
  const layout = useDefaultLayout({ id: 'query', storage: layoutStorage })
  const onReady = useCallback((e: MonacoEditor | null) => {
    editor.current = e
  }, [])

  useEffect(
    () =>
      onCommand((cmd) => {
        const ed = editor.current
        const model = ed?.getModel()
        if (!ed || !model) return
        const sel = ed.getSelection()
        const selected = sel && !sel.isEmpty() ? sel : null
        const part = () =>
          selected
            ? { sql: model.getValueInRange(selected), base: model.getOffsetAt(selected.getStartPosition()) }
            : null
        const cursor = model.getOffsetAt(ed.getPosition() ?? { lineNumber: 1, column: 1 })
        switch (cmd) {
          case 'run-current':
          case 'explain': {
            const explain = cmd === 'explain'
            const p = part()
            if (p) void runner.run(p.sql, p.base, { explain }, model.getValue())
            else void runner.run(model.getValue(), 0, { cursor, explain })
            break
          }
          case 'run-all': {
            const p = part() ?? { sql: model.getValue(), base: 0 }
            void runner.run(p.sql, p.base, {}, model.getValue())
            break
          }
          case 'cancel':
            runner.cancel()
            break
          case 'format': {
            if (!profile) return
            const range = selected ?? model.getFullModelRange()
            try {
              const text = formatSql(model.getValueInRange(range), profile.driver, rt.session?.variant)
              ed.executeEdits('format', [{ range, text }])
            } catch (e) {
              message.error(`无法格式化：${e instanceof Error ? e.message : String(e)}`)
            }
            break
          }
        }
      }),
    [runner, profile, rt.session?.variant, message],
  )

  // 协议层面的错误（如上一段还没执行完）用全局提示（M1 设计文档第 5 节），同时记在"消息"里
  const protocolError = rt.exec.protocolError
  useEffect(() => {
    if (protocolError) message.error(errorText(protocolError))
  }, [protocolError, message])

  const marks = useMemo(() => errorMarks(rt.exec, rt.source), [rt.exec, rt.source])
  const confirm = !rt.running && !rt.declined ? rt.exec.confirm : null
  const sub = rt.submission

  return (
    <div className="query">
      <Toolbar tab={tab} rt={rt} runner={runner} profiles={profiles} />
      {!profile && <Alert type="error" banner message="这个标签页的连接已被删除，请在工具栏中换一个连接" />}
      {rt.session?.broken && (
        <Alert
          type="warning"
          banner
          message="与数据库的连接已断开，未提交的修改已丢失"
          action={
            <Button size="small" onClick={() => runner.reconnect().catch((e: Error) => message.error(e.message))}>
              重新连接
            </Button>
          }
        />
      )}
      <Group orientation="vertical" className="grow" {...layout}>
        <Panel id="editor" minSize={80}>
          <SqlEditor
            path={tab.id}
            language={profile ? editorLanguage(profile.driver) : 'sql'}
            defaultValue={tab.sql}
            marks={marks}
            onChange={(v) => useTabs.getState().setSql(tab.id, v)}
            onReady={onReady}
          />
        </Panel>
        <Separator className="splitter-h" />
        <Panel id="results" defaultSize="40%" minSize={60}>
          <ResultsPanel rt={rt} onFetchMore={() => runner.fetchMore()} />
        </Panel>
      </Group>
      <ConfirmDialog
        open={confirm !== null}
        sql={confirm && sub ? sub.sql.slice(confirm.start - sub.base, confirm.end - sub.base) : ''}
        reason={confirm?.reason ?? ''}
        onConfirm={() => void runner.confirm()}
        onCancel={() => runner.dismissConfirm()}
      />
    </div>
  )
}
```

```tsx
// 新建、关闭查询标签页。菜单、快捷键、对象树、标签栏共用；不引入 Monaco，外壳可以直接使用。

import { App, Button } from 'antd'
import { useQuery } from '@tanstack/react-query'

import { useApi } from '../../app/context'
import { useUi } from '../../app/ui'
import { disposeRunner, useRuntime } from './runner'
import { useTabs } from './tabs'

export function useQueryActions() {
  const api = useApi()
  const { modal } = App.useApp()
  const { data: profiles } = useQuery({ queryKey: ['profiles'], queryFn: () => api.profiles() })

  /** 新建查询：用给定的连接，否则依次用对象树中选中的、当前标签的、第一个连接；一个连接都没有时打开新建连接。 */
  const newQuery = (profileId?: string) => {
    const { tabs, active } = useTabs.getState()
    const id =
      profileId ??
      useUi.getState().selectedProfile ??
      tabs.find((t) => t.id === active)?.profileId ??
      profiles?.[0]?.id
    if (id) useTabs.getState().open(id)
    else useUi.getState().openConnection(null)
  }

  /** 关闭标签页：有未提交的事务时先问提交还是回滚。 */
  const closeQuery = (tabId: string) => {
    const tab = useTabs.getState().tabs.find((t) => t.id === tabId)
    if (!tab) return
    const finish = async (commit: boolean) => {
      await disposeRunner(tabId, commit)
      useTabs.getState().close(tabId)
    }
    if (!useRuntime.getState().tabs[tabId]?.session?.in_transaction) {
      void finish(false)
      return
    }
    const m = modal.confirm({
      title: `${tab.title} 有未提交的事务`,
      content: '关闭前提交还是回滚这些修改？',
      okText: '提交并关闭',
      cancelText: '取消',
      onOk: () => finish(true),
      footer: (_, { OkBtn, CancelBtn }) => (
        <>
          <CancelBtn />
          <Button
            danger
            onClick={() => {
              m.destroy()
              void finish(false)
            }}
          >
            回滚并关闭
          </Button>
          <OkBtn />
        </>
      ),
    })
  }

  const closeActive = () => {
    const active = useTabs.getState().active
    if (active) closeQuery(active)
  }

  return { newQuery, closeQuery, closeActive }
}
```

```tsx
/** 全局快捷键（M1 设计文档 1.3 节）。焦点在编辑器里时由 Monaco 的动作处理，这里处理焦点在别处的情况。 */
function shortcut(e: KeyboardEvent): EditorCommand | 'new-connection' | 'new-query' | 'close-tab' | null {
  const ctrl = e.ctrlKey || e.metaKey
  const key = e.key.toLowerCase()
  if (ctrl && !e.shiftKey && !e.altKey && key === 'n') return 'new-connection'
  if (ctrl && !e.shiftKey && !e.altKey && key === 't') return 'new-query'
  if (ctrl && !e.shiftKey && !e.altKey && key === 'w') return 'close-tab'
  if (ctrl && key === 'enter') return 'run-current'
  if (!ctrl && !e.altKey && key === 'f5') return 'run-all'
  if ((ctrl && key === 'pause') || (e.metaKey && key === '.')) return 'cancel'
  if (e.shiftKey && e.altKey && key === 'f') return 'format'
  return null
}
```

````markdown
## 协议的变化

接口说明见 `docs/notes/m0-4b-api.md`。WebSocket 的 `exec` 消息新增三个可选字段：
- `continue_on_error`：某条语句失败或被策略拒绝后，继续执行后面的。需要确认、被取消、连接断开时仍然停下。
- `cursor`：UTF-16 偏移，只执行它所在的那条语句。
  - 落在两条语句之间时取前一条；在第一条之前时取第一条。
  - 事件里的偏移仍是相对于整段提交的 SQL。
- `explain`：只生成执行计划，只能针对一条语句（多条时报 `invalid`）。语句按方言生成：
  - MySQL 系、PG 系：`EXPLAIN …`；
  - 达梦：`EXPLAIN FOR …`，返回计划表的各行。`EXPLAIN …` 经 DPI 执行不返回结果集（实测）。
  - Oracle：先 `EXPLAIN PLAN FOR …`，再 `SELECT … FROM TABLE(DBMS_XPLAN.DISPLAY())`。
  - 这些语句都按 `EXPLAIN` 分类，不会触发生产确认。

`confirmed` 的含义改为只对第一条语句有效：确认后从那条重新提交，后面需要确认的语句仍会停下，每条单独确认。

事件的变化：
- `statement_start` 多了 `total`，表示这次提交共几条。
  - 前面语句的结果没取完，后面的语句就执行不了。
  - 所以前端对非最后一条语句自动续取到底（受行数上限约束）；最后一条等表格滚动到底部再续取。
- `failed` 多了可选的 `offset`：数据库报告的出错位置，换算成 UTF-16 偏移。
- `confirm_required` 多了 `end`，确认框据此显示整条语句。

会话：
- 会话信息多了 `schema`，即当前 schema（MySQL 系为当前库）。
  - 打开会话时读取。
  - 执行 `USE` / `SET` / `ALTER` 开头的语句后重新读取。
- 新增 `POST /api/sessions/{id}/schema`，请求体 `{"schema": "…"}`，用于切换。

## 出错位置

`Error::Sql::position` 是语句内的字符偏移，从 1 开始。各库的来源都实测过（`SELECT '数洞', FROM t` 在 5 个库上都指向 `FROM`）：

| 库 | 来源 |
|---|---|
| PostgreSQL 系 | 协议直接给字符偏移 |
| Oracle | ODPI 给的是字节偏移，在执行处用 `Error::position_bytes_to_chars` 换成字符 |
| MySQL 系 | 错误信息 `… near 'FROM t' at line 1`：在该行起找 near 后面原文的第一行 |
| 达梦 | 错误信息 `第 1 行, 第 17 列[FROM]附近出现错误`：列号在第一行和其他行之间差 1，不可靠，所以在该行里找方括号中的词，取离列号最近的一处 |

没有位置时（如达梦的"无效的列名"、语义错误），编辑器标出整条语句。

## 前端要点

- **Monaco**：从本地打包加载（`loader.config({ monaco })`），只用编辑器自己的 worker，离线可用。工作区懒加载。
  - 每个标签一个模型（以标签 id 为路径），切换标签时保留撤销历史；关闭标签时删除模型。
  - 快捷键注册为编辑器动作；焦点不在编辑器时由外壳的全局快捷键处理。F5 在浏览器里也被拦下，不会刷新页面。
- **执行当前语句**：有选中时执行选中部分，否则把全文和光标位置交给后端，由后端按方言拆分选出那条，前端不自己拆分。
- **偏移**：执行选中部分、确认后续跑时提交的只是全文的一段。执行器收到事件后加上这段的起点，状态里存的都是全文中的偏移。
  - 执行时保存当时的全文：错误标记和失败语句的显示都按它取文本，之后的编辑不影响。
- **行数上限**：设置项 `ui.exec.row-cap`，默认 10 万行，可在设置对话框中修改。
  - 达到上限时取消这条语句（服务端在等 `fetch_more`，不取消的话会话一直忙），后面的语句也不再执行，消息里说明。
- **确认框**：取消后脚本已停下，消息中写明这条及后面的语句没有执行。
- **关闭标签**：会话有未提交的事务时，询问提交、回滚或取消。
- **结果表格**：行号、数值右对齐、NULL 灰显。
  - 列宽在第一批行到达时估计。
  - 双击或回车打开单元格查看（文本 / JSON / 十六进制）。
  - 选中区域可用 Ctrl+C 复制，按制表符分隔。
- **主题**：Monaco 的两套配色和表格的颜色都取自外壳的 CSS 变量，随主题切换。
````

```markdown
    - `continue_on_error`：失败后继续执行；

    - `explain`：只生成执行计划。
```

```markdown
| `statement_start` | `index, total, start, end, info` | 开始执行第 `index` 条语句，这次提交共 `total` 条。`start`/`end` 是它在提交的 SQL 中的 UTF-16 偏移（可直接用于 Monaco）。`info` 是语句分析结果 |
```
